cmake_minimum_required(VERSION 3.20)
project(EuroSkyConnect VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(eurosky_core src/eurosky.cpp)
target_include_directories(eurosky_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(eurosky_core PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(eurosky_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(eurosky_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(eurosky_core PRIVATE winhttp)
endif()

add_executable(eurosky_cli src/main.cpp)
target_link_libraries(eurosky_cli PRIVATE eurosky_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(eurosky_tests tests/test_main.cpp)
  target_link_libraries(eurosky_tests PRIVATE eurosky_core)
  target_compile_definitions(eurosky_tests PRIVATE EUROSKY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME eurosky_tests COMMAND eurosky_tests)
endif()
